## Architecture Overview

### System Architecture

```
User Input
    ↓
Preflight Checks (Git, GitHub, gh CLI)
    ↓
Feature Recognition (Auto-detect or ask)
    ↓
Techspec Resolution (4 priority levels)
    ↓
Branch Management (Check & switch)
    ↓
Mode Determination (INITIALIZE or RESUME)
    ↓
    ├─→ INITIALIZE
    │     ├─ Analyze Techspec (Claude)
    │     ├─ Map Agents (Claude)
    │     ├─ Generate plan.md
    │     ├─ Create GitHub Issues
    │     ├─ Generate TODO.md
    │     └─ Start Phase 1
    │
    └─→ RESUME
          ├─ Sync TODO ↔ GitHub
          ├─ Show Status Report
          └─ Continue Execution
    ↓
Parallel Execution (Multiple agents)
    ↓
Validation & Commit
    ↓
Sync TODO.md ↔ GitHub Issues
    ↓
Exit (Force sync, final report)
```

### Directory Structure

```
project/
├── .git/
│   └── config                    # Must have remote
├── docs/
│   └── features/                 # Feature-based organization
│       ├── toss-integration/
│       │   ├── techspec.md      # INPUT (flexible naming)
│       │   ├── plan.md          # Generated by skill
│       │   └── TODO.md          # Generated & synced
│       └── stripe-integration/
│           └── ...
└── .github/
    └── (issues created via gh CLI)
```

```typescript
function preflightChecks(): boolean {
  console.log('🔍 Running preflight checks...\n');

  // 1. Git repository check
  try {
    execSync('git status', { stdio: 'ignore' });
    console.log('✅ Git repository');
  } catch {
    console.error('❌ Not a git repository!');
    console.error('Run: git init');
    return false;
  }

  // 2. Git remote check (CRITICAL!)
  try {
    const remote = execSync('git remote -v').toString();

    if (!remote || remote.trim() === '') {
      console.error('❌ Git remote not configured!');
      console.error('\nThis skill requires GitHub integration.');
      console.error('\nAdd remote:');
      console.error('  git remote add origin https://github.com/user/repo.git');
      return false;
    }

    console.log('✅ Git remote configured');
    console.log(`   ${remote.split('\n')[0]}`);

  } catch {
    console.error('❌ Failed to check git remote');
    return false;
  }

  // 3. GitHub CLI check
  try {
    execSync('gh --version', { stdio: 'ignore' });
    console.log('✅ GitHub CLI (gh) installed');
  } catch {
    console.error('❌ GitHub CLI not found!');
    console.error('\nInstall: https://cli.github.com/');
    return false;
  }

  // 4. GitHub authentication check
  try {
    execSync('gh auth status', { stdio: 'ignore' });
    console.log('✅ GitHub authenticated\n');
  } catch {
    console.error('❌ GitHub not authenticated!');
    console.error('Run: gh auth login');
    return false;
  }

  return true;
}
```

**Exit Points**:
- No git repo → Exit
- No git remote → Exit
- No gh CLI → Exit
- Not authenticated → Exit

### Step 1: Feature Recognition

**Purpose**: Determine which feature to work on.

```typescript
function detectFeature(userInput: string): string {
  // Priority 1: Explicit feature name in input
  // "Start stripe-integration feature"
  // "Continue toss-integration"
  const explicit = extractExplicitFeature(userInput);
  if (explicit) {
    console.log(`✅ Feature: ${explicit}`);
    return explicit;
  }

  // Priority 2: Match against existing features
  const features = listDirectories('docs/features/');
  // → ['toss-integration', 'stripe-integration']

  const matched = features.find(feature => {
    const keywords = feature.split('-');
    return keywords.some(keyword =>
      userInput.toLowerCase().includes(keyword.toLowerCase())
    );
  });

  if (matched) {
    console.log(`✅ Feature detected: ${matched}`);
    return matched;
  }

  // Priority 3: Ask user
  const suggested = generateFeatureName(userInput);

  const answer = askUser({
    question: "Feature 이름을 정해주세요",
    defaultValue: suggested,
    options: [
      { label: suggested, description: "추천 이름" },
      { label: "직접 입력", description: "Custom name" }
    ]
  });

  return answer === "직접 입력" ? askForCustomName() : suggested;
}

function generateFeatureName(userInput: string): string {
  // Extract keywords and convert to kebab-case
  // "Stripe 결제 연동해줘" → "stripe-integration"
  // "환불 기능 추가" → "refund-feature"

  const keywords = extractKeywords(userInput);
  return kebabCase(keywords.join(' '));
}

function extractExplicitFeature(input: string): string | null {
  // Pattern matching for explicit feature names
  const patterns = [
    /(\w+-\w+) feature/i,        // "stripe-integration feature"
    /feature[:\s]+(\w+-\w+)/i,   // "feature: toss-integration"
    /continue\s+(\w+-\w+)/i      // "continue toss-integration"
  ];

  for (const pattern of patterns) {
    const match = input.match(pattern);
    if (match) return match[1];
  }

  return null;
}
```

```typescript
function checkAndSwitchBranch(feature: string): boolean {
  const current = execSync('git branch --show-current').toString().trim();
  const expected = `feature/${feature}`;

  if (current === expected) {
    console.log(`✅ Branch: ${expected}`);
    return true;
  }

  console.log(`\n⚠️ Branch Mismatch`);
  console.log(`  Current: ${current}`);
  console.log(`  Expected: ${expected}\n`);

  const answer = askUser({
    question: "브랜치를 전환할까요?",
    options: [
      "전환하고 계속",
      "현재 브랜치에서 계속",
      "취소"
    ]
  });

  if (answer === "취소") {
    return false;
  }

  if (answer === "전환하고 계속") {
    const branchExists = execSync('git branch --list')
      .toString()
      .includes(expected);

    if (branchExists) {
      execSync(`git checkout ${expected}`);
      console.log(`✅ Switched to ${expected}`);
    } else {
      execSync(`git checkout -b ${expected}`);
      console.log(`✅ Created and switched to ${expected}`);
    }
  } else {
    console.log(`⚠️ Continuing on ${current}`);
  }

  return true;
}
```

```typescript
async function createGitHubIssues(analysis, feature: string) {
  console.log(`\n📋 Creating GitHub Issues...\n`);

  let issueNumber = await getNextIssueNumber();
  const issueMap = {};

  for (const phase of analysis.phases) {
    for (const task of phase.tasks) {
      const issueBody = generateIssueBody(task, feature, phase);

      const result = execSync(`gh issue create \
        --title "[${feature}][Phase ${phase.number}] ${task.name}" \
        --label "feature:${feature}" \
        --label "branch:feature/${feature}" \
        --label "phase:${phase.number}" \
        --label "tdd" \
        --label "status:pending" \
        --body "${escapeForShell(issueBody)}"
      `).toString();

      // Extract issue number from output
      const match = result.match(/#(\d+)/);
      if (match) {
        issueMap[task.name] = parseInt(match[1]);
        console.log(`  ✅ #${match[1]} ${task.name}`);
      }
    }
  }

  console.log(`\n✅ Created ${Object.keys(issueMap).length} issues\n`);

  return issueMap;
}
```

## Synchronization

### TODO.md ↔ GitHub Issues

**Philosophy**: GitHub Issues is the source of truth, TODO.md is a fast local view.

```typescript
function syncTodoWithGitHub(feature: string) {
  console.log('🔄 Syncing TODO.md with GitHub Issues...');

  // 1. Get GitHub Issues
  const ghIssues = getGitHubIssues(feature);

  // 2. Read current TODO.md
  const todoPath = `docs/features/${feature}/TODO.md`;
  let content = readFile(todoPath);

  // 3. Update each task based on GitHub state
  ghIssues.forEach(issue => {
    const issueNum = issue.number;
    const newState = issue.state === 'closed' ? 'x' : ' ';

    // Pattern: - [ ] Task name → [#123]
    const pattern = new RegExp(
      `- \\[(.)\\] (.+?) → \\[#${issueNum}\\]`,
      'g'
    );

    content = content.replace(pattern, (match, oldState, taskName) => {
      return `- [${newState}] ${taskName} → [#${issueNum}]`;
    });
  });

  // 4. Update phase status
  // If all tasks in a phase are closed, mark phase as ✅
  const phasePattern = /## Phase (\d+): (.+?) (⏳|✅) \((\d+)\/(\d+)\)/g;

  content = content.replace(phasePattern, (match, num, name, status, done, total) => {
    const phaseTasks = ghIssues.filter(i =>
      i.labels.some(l => l.name === `phase:${num}`)
    );
    const phaseClosed = phaseTasks.filter(i => i.state === 'closed').length;
    const phaseTotal = phaseTasks.length;
    const phaseStatus = phaseClosed === phaseTotal ? '✅' : '⏳';

    return `## Phase ${num}: ${name} ${phaseStatus} (${phaseClosed}/${phaseTotal})`;
  });

  // 5. Update overall progress
  const progress = calculateProgress(content);
  content = content.replace(
    /\*\*Overall Progress\*\*: \d+% \(\d+\/\d+ tasks\)/,
    `**Overall Progress**: ${progress.percentage}% (${progress.completed}/${progress.total} tasks)`
  );

  // 6. Update sync timestamp
  content = content.replace(
    /Last synced with GitHub: .+/,
    `Last synced with GitHub: ${new Date().toISOString()}`
  );

  content = content.replace(/Open Issues: \d+/, `Open Issues: ${progress.total - progress.completed}`);
  content = content.replace(/Closed Issues: \d+/, `Closed Issues: ${progress.completed}`);

  // 7. Write back
  writeFile(todoPath, content);

  console.log(`✅ TODO.md synced (${progress.completed}/${progress.total})`);
}

function getGitHubIssues(feature: string) {
  const json = execSync(
    `gh issue list --label "feature:${feature}" --json number,state,labels,title --limit 100`
  ).toString();

  return JSON.parse(json);
}

function calculateProgress(todoContent: string) {
  const tasks = todoContent.match(/- \[.\]/g) || [];
  const total = tasks.length;
  const completed = tasks.filter(t => t.includes('[x]')).length;

  return {
    completed,
    total,
    percentage: Math.round((completed / total) * 100)
  };
}
```

```typescript
async function executePhaseInParallel(feature: string, phaseNumber: number) {
  const plan = readPlanMd(feature);
  const phase = plan.phases[phaseNumber - 1];

  console.log(`🚀 Starting ${phase.name} (${phase.tasks.length} tasks in parallel)\n`);

  // Get issue bodies for all tasks
  const issueBodies = await Promise.all(
    phase.tasks.map(task =>
      execSync(`gh issue view ${task.issueNumber} --json body -q .body`).toString()
    )
  );

  // Launch all agents in SINGLE message
  const agentPromises = phase.tasks.map((task, i) => {
    return Task({
      subagent_type: task.agent,
      prompt: `
You are executing a TDD task for feature: ${feature}

${issueBodies[i]}

CRITICAL:
1. Follow TDD: RED → GREEN → REFACTOR

4. Update TODO.md
      `
    });

  // Wait for all
  await Promise.all(agentPromises);

  // Validate

    tests: execSync('npm test').exitCode === 0,
    build: execSync('npm run build').exitCode === 0,
    lint: execSync('npm run lint').exitCode === 0
  };

  if (!Object.values(validation).every(v => v)) {

  // Phase commit
  const issueRefs = phase.tasks.map(t => `#${t.issueNumber}`).join(', ');
```

```typescript
  // 1. Check uncommitted changes
  const uncommitted = execSync('git status --porcelain').toString();

  if (uncommitted.length > 0) {
    console.warn('⚠️ Uncommitted changes:');
    console.log(uncommitted);

    const answer = askUser({
      question: 'Commit changes?',
      options: ['Yes', 'No']
    });

    if (answer === 'Yes') {
      const message = askUser({
        question: 'Commit message:',
        defaultValue: 'wip: Work in progress'
      });

      execSync(`git add .`);
      execSync(`git commit -m "${message}"`);
      console.log('✅ Changes committed');
    }
  }

  // 2. Force sync
  const ghIssues = getGitHubIssues(feature);
  const todoContent = readFile(`docs/features/${feature}/TODO.md`);

  const ghClosed = ghIssues.filter(i => i.state === 'closed').length;
  const todoChecked = (todoContent.match(/\[x\]/g) || []).length;

  if (ghClosed !== todoChecked) {
    console.warn(`⚠️ Sync mismatch! GitHub: ${ghClosed}, TODO: ${todoChecked}`);
    console.log('Force syncing...');
    forceSyncTodoWithGitHub(feature);
    console.log('✅ Synced');
  } else {
    console.log('✅ Already synced');
  }

  // 3. Final report
  const progress = calculateProgress(todoContent);

  console.log(`
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
✅ Session Complete: ${feature}
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

📊 Progress: ${progress.percentage}% (${progress.completed}/${progress.total})
🔗 GitHub: https://github.com/user/repo/issues?q=label:feature:${feature}
🌿 Branch: feature/${feature}

To resume:
  "Resume ${feature} feature"

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  `);
}
```
